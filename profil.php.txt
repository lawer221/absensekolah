<?php

session_start();

if(!isset($_SESSION['siswa'])){

    header("Location:index.php");
    exit;

}

$file = "data/siswa.json";

$siswa = json_decode(
    file_get_contents($file),
    true
);

$nis = $_SESSION['siswa'];

if(isset($_POST['upload'])){

    if(isset($_FILES['foto']) &&
       $_FILES['foto']['error'] === 0){

        if(!is_dir("uploads")){
            mkdir("uploads",0755,true);
        }

        $ext = strtolower(
            pathinfo(
                $_FILES['foto']['name'],
                PATHINFO_EXTENSION
            )
        );

        $allowed = [
            "jpg",
            "jpeg",
            "png",
            "webp"
        ];

        if(in_array($ext,$allowed)){

            $namaFoto =
            $nis . "_" . time() . "." . $ext;

            move_uploaded_file(
                $_FILES['foto']['tmp_name'],
                "uploads/".$namaFoto
            );

            foreach($siswa as &$s){

                if($s['nis'] === $nis){

                    $s['foto'] =
                    $namaFoto;

                    break;
                }
            }

            file_put_contents(
                $file,
                json_encode(
                    $siswa,
                    JSON_PRETTY_PRINT
                )
            );

            header("Location:siswa.php");
            exit;
        }
    }
}
