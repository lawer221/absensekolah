<?php

session_start();

if(!isset($_SESSION['siswa'])){

    header("Location:index.php");
    exit;

}

$nis = $_SESSION['siswa'];

$file = "data/absensi.json";

if(!file_exists($file)){
    file_put_contents($file,"[]");
}

$absensi = json_decode(
    file_get_contents($file),
    true
);

$pesan = "";
$tipe = "";

if(isset($_POST['absen'])){

    $tanggal = date("Y-m-d");

    $sudah = false;

    foreach($absensi as $a){

        if(
            $a['nis'] === $nis &&
            $a['tanggal'] === $tanggal
        ){

            $sudah = true;
            break;
        }
    }

    if($sudah){

        $pesan =
        "Kamu sudah absen hari ini.";

        $tipe = "error";

    }else{

        $absensi[] = [

            "nis" => $nis,
            "tanggal" => $tanggal,
            "jam" => date("H:i:s"),
            "status" => "Hadir",
            "lokasi" =>
            $_POST['lokasi'] ?? "Tidak tersedia"

        ];

        file_put_contents(
            $file,
            json_encode(
                $absensi,
                JSON_PRETTY_PRINT
            )
        );

        $pesan =
        "✅ Absensi berhasil dan sudah masuk ke Admin.";

        $tipe = "success";
    }
}

function jam(){

    let d = new Date();

    document.getElementById("clock")
    .innerText =
    d.toLocaleTimeString("id-ID");

}

navigator.geolocation.getCurrentPosition(

.value =
