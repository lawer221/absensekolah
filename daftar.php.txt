<?php
session_start();

$file = "data/siswa.json";

if(!is_dir("data")){
    mkdir("data",0755,true);
}

if(!file_exists($file)){
    file_put_contents($file,"[]");
}

$siswa = json_decode(
    file_get_contents($file),
    true
);

$error = "";

if(isset($_POST['daftar'])){

    $nis = trim($_POST['nis']);
    $nama = trim($_POST['nama']);
    $kelas = trim($_POST['kelas']);
    $jk = trim($_POST['jk']);
    $password = $_POST['password'];

    foreach($siswa as $data){

        if($data['nis'] === $nis){

            $error = "NIS sudah terdaftar.";
            break;
        }
    }

    if($error === ""){

        $siswa[] = [

            "nis" => $nis,
            "nama" => $nama,
            "kelas" => $kelas,
            "jk" => $jk,
            "password" => $password,
            "foto" => ""

        ];

        file_put_contents(
            $file,
            json_encode(
                $siswa,
                JSON_PRETTY_PRINT
            )
        );

        header("Location:index.php");
        exit;
    }
}
